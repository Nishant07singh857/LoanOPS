   123→function timeAgo(dateStr: string): string {
   124→  const d = new Date(dateStr)
   125→  const diff = Date.now() - d.getTime()
   126→  const sec = Math.floor(diff / 1000)
   127→  if (sec < 60) return `${sec}s ago`
   128→  const min = Math.floor(sec / 60)
   129→  if (min < 60) return `${min}m ago`
   130→  const hr = Math.floor(min / 60)
   131→  if (hr < 24) return `${hr}h ago`
   132→  const day = Math.floor(hr / 24)
   133→  if (day < 7) return `${day}d ago`
   134→  return d.toLocaleDateString('en-IN', { month: 'short', day: 'numeric' })
   135→}
